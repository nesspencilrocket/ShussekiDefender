    private void Log(string message,string stack,LogType type)
    {
        if(type==LogType.Error||type==LogType.Exception){errors++;File.AppendAllText(output+"/errors.txt",message+"\n"+stack+"\n");}
        else if(type==LogType.Warning)File.AppendAllText(output+"/warnings.txt",message+"\n");
    }
    private void Check(bool condition,string label)
    {
        if(condition)passed++;else failed++;
        File.AppendAllText(output+"/report.txt",(condition?"PASS ":"FAIL ")+label+"\n");
    }
    private void Attempt(Action action,string label)
    {try{action();Check(true,label);}catch(Exception error){Check(false,label+" EXCEPTION "+error.GetBaseException().GetType().Name);}}
    private static T Get<T>(object host,string name)=> (T)host.GetType().GetField(name,BindingFlags.NonPublic|BindingFlags.Instance).GetValue(host);
    private static void Set(object host,string name,object value)=>host.GetType().GetField(name,BindingFlags.NonPublic|BindingFlags.Instance).SetValue(host,value);
    private static void Call(object host,string name)=>host.GetType().GetMethod(name,BindingFlags.NonPublic|BindingFlags.Instance).Invoke(host,null);
    private void Resize(int width,int height)
    {
        string key=width+"x"+height;
        if(!sizeIndexes.ContainsKey(key))
        {
            Assembly editor=typeof(EditorWindow).Assembly;
            Type sizes=editor.GetType("UnityEditor.GameViewSizes");
            object instance=sizes.GetProperty("instance",BindingFlags.Public|BindingFlags.Static|BindingFlags.FlattenHierarchy).GetValue(null);
            group=sizes.GetMethod("GetGroup").Invoke(instance,new[]{Enum.Parse(editor.GetType("UnityEditor.GameViewSizeGroupType"),"Standalone")});
            object size=Activator.CreateInstance(editor.GetType("UnityEditor.GameViewSize"),Enum.Parse(editor.GetType("UnityEditor.GameViewSizeType"),"FixedResolution"),width,height,"CommonQa-"+key);
            group.GetType().GetMethod("AddCustomSize").Invoke(group,new[]{size});
            sizeIndexes[key]=(int)group.GetType().GetMethod("GetBuiltinCount").Invoke(group,null)+(int)group.GetType().GetMethod("GetCustomCount").Invoke(group,null)-1;
        }
        selection.SetValue(view,sizeIndexes[key]);view.Repaint();
    }
    private IEnumerator Ready()

    private IEnumerator Capture(string name)
    {yield return new WaitForEndOfFrame();ScreenCapture.CaptureScreenshot(output+"/"+name+".png");yield return null;}
    private void Fit(TextMeshProUGUI text,string label)
    {
        if(text==null||!text.gameObject.activeInHierarchy||string.IsNullOrWhiteSpace(text.text))return;
        text.ForceMeshUpdate();Rect r=text.rectTransform.rect;Vector4 m=text.margin;Bounds b=text.textBounds;
        Check(!text.isTextOverflowing&&b.min.x>=r.xMin+m.x-2&&b.max.x<=r.xMax-m.z+2
            &&b.min.y>=r.yMin+m.w-2&&b.max.y<=r.yMax-m.y+2,label+" fits text="+text.text.Replace("\n","/")+" size="+text.fontSize+" lines="+text.textInfo.lineCount);
    }
    private void Visible(TextMeshProUGUI text,string label)
    {
        if(text==null||!text.gameObject.activeInHierarchy)return;
        text.ForceMeshUpdate();Bounds bounds=text.textBounds;Rect pixels=Camera.main.pixelRect;
        Vector2 min=RectTransformUtility.WorldToScreenPoint(null,text.transform.TransformPoint(bounds.min));
        Vector2 max=RectTransformUtility.WorldToScreenPoint(null,text.transform.TransformPoint(bounds.max));
        Check(min.x>=pixels.xMin-2&&max.x<=pixels.xMax+2&&min.y>=pixels.yMin-2&&max.y<=pixels.yMax+2,label+" stays inside gameplay viewport");
    }
    private void Buttons(string label)
    {
        foreach(Button button in FindObjectsByType<Button>(FindObjectsSortMode.None))
        {
            Canvas canvas=button.GetComponentInParent<Canvas>();
            if(!button.IsInteractable()||canvas==null||canvas.renderMode!=RenderMode.ScreenSpaceOverlay)continue;
            RectTransform rt=(RectTransform)button.transform;
            Vector2 point=RectTransformUtility.WorldToScreenPoint(null,rt.TransformPoint(rt.rect.center));
            var hits=new List<RaycastResult>();
            EventSystem.current.RaycastAll(new PointerEventData(EventSystem.current){position=point},hits);
            Check(hits.Count>0&&hits[0].gameObject.GetComponentInParent<Button>()==button,label+" button raycast "+button.name);
        }
    }

    private Node HitNode(Vector3 world)
    {
        var hits=new List<RaycastResult>();
        EventSystem.current.RaycastAll(new PointerEventData(EventSystem.current){position=Camera.main.WorldToScreenPoint(world)},hits);
        return hits.Count==0?null:hits[0].gameObject.GetComponentInParent<Node>();
    }
    private IEnumerator NodeChecks()
    {
        Node target=FindObjectsByType<Node>(FindObjectsSortMode.None).Single(n=>n.UnlockOrder==1);
        SpriteRenderer sprite=target.GetComponent<SpriteRenderer>();
        Node[] unlocked=FindObjectsByType<Node>(FindObjectsSortMode.None).Where(n=>n.IsUnlocked).ToArray();
        FindFirstObjectByType<Spawner>().StopAllCoroutines();
        foreach(Enemy e in FindObjectsByType<Enemy>(FindObjectsSortMode.None))e.gameObject.SetActive(false);
        int[] widths={1920,1280,1024};int[] heights={1080,800,768};
        File.AppendAllText(output+"/report.txt","TARGET "+target.name+" world="+target.transform.position+" visual="+sprite.bounds.size+"\n");
        for(int i=0;i<widths.Length;i++)
        {
            Resize(widths[i],heights[i]);for(int frame=0;frame<4;frame++)yield return null;
            string label=widths[i]+"x"+heights[i];
            foreach(float y in new[]{-.45f,-.225f,0f,.225f,.45f})
                foreach(float x in new[]{-.45f,-.225f,0f,.225f,.45f})
                {
                    Vector3 world=sprite.bounds.center+new Vector3(sprite.bounds.size.x*x,sprite.bounds.size.y*y,0);
                    Check(HitNode(world)==target,label+" target node hit x="+x+" y="+y);
                }
            foreach(Node n in unlocked)Check(HitNode(n.GetComponent<SpriteRenderer>().bounds.center)==n,label+" other node center "+n.UnlockOrder);
            Node neighbor=unlocked.OrderBy(n=>Vector3.Distance(n.transform.position,target.transform.position)).Skip(1).First();
            Check(HitNode((target.transform.position+neighbor.transform.position)/2f)==null,label+" gap between nodes stays nonclickable");
            Vector3 top=sprite.bounds.center+Vector3.up*(sprite.bounds.size.y*.45f);
            var hits=new List<RaycastResult>();
            EventSystem.current.RaycastAll(new PointerEventData(EventSystem.current){position=Camera.main.WorldToScreenPoint(top)},hits);
            if(hits.Count>0)ExecuteEvents.Execute(hits[0].gameObject,new PointerEventData(EventSystem.current){button=PointerEventData.InputButton.Left},ExecuteEvents.pointerClickHandler);
            Check(Get<GameObject>(UIManager.instance,"weaponShopPanel").activeSelf,label+" upper edge opens shop");
            UIManager.instance.CloseTurretShopPanel();
        }
        int selected=0;Action<Node> handler=n=>selected++;Node.OnNodeSelected+=handler;
        FindObjectsByType<Node>(FindObjectsSortMode.None).Single(n=>n.UnlockOrder==5).SelectNode();
        Check(selected==0,"locked nodes remain locked");
        GameManager.Instance.GameOver();target.SelectNode();
        Check(selected==0&&!Get<GameObject>(UIManager.instance,"weaponShopPanel").activeSelf,"result still blocks node selection");
        Node.OnNodeSelected-=handler;
        if(run=="node-manual")
        {
            GameManager.Instance.OnRetryButtonClicked();yield return Ready();GameSpeed.Pause();FindFirstObjectByType<Spawner>().StopAllCoroutines();
            foreach(Enemy e in FindObjectsByType<Enemy>(FindObjectsSortMode.None))e.gameObject.SetActive(false);
            Resize(1920,1080);for(int frame=0;frame<3;frame++)yield return null;
            yield return Capture("node-fixed");
        }
        RestorePrefs();File.AppendAllText(output+"/report.txt","NODES_DONE passed="+passed+" failed="+failed+" errors="+errors+"\n");
    }

        }
